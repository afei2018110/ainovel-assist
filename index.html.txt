<script>
let agents = [];
function addAgent(){
  let name = prompt("输入AI角色名称","总编");
  let prompt = prompt("输入角色设定","统筹全局，分配任务，把控剧情");
  if(!name||!prompt)return;
  agents.push({name,prompt});
  renderAgent();
}
function renderAgent(){
  let html = "";
  agents.forEach((item,idx)=>{
    html += `<div class="agent-item">
      <span>${item.name}</span>
      <span style="color:#999;font-size:12px">${item.prompt}</span>
      <button onclick="delAgent(${idx})" style="background:#e53935;padding:4px 8px;font-size:12px">删除</button>
    </div>`
  })
  document.getElementById("agentList").innerHTML = html;
}
function delAgent(idx){agents.splice(idx,1);renderAgent()}
function clearChat(){document.getElementById("chatBox").innerHTML=""}

async function sendTask(){
  let key = document.getElementById("apiKey").value;
  let task = document.getElementById("userInput").value.trim();
  let model = document.getElementById("modelType").value;
  if(!key||!task||agents.length===0)return alert("请填写密钥、任务、添加AI角色");
  let chat = document.getElementById("chatBox");
  chat.innerHTML += `<div class="msg user-msg">用户：${task}</div>`;
  document.getElementById("userInput").value="";

  for(let ai of agents){
    chat.innerHTML += `<div class="msg ai-msg">【${ai.name}】正在处理中...</div>`;
    let res = await fetchAi(key,model,task,ai.prompt);
    chat.innerHTML += `<div class="msg ai-msg">【${ai.name}】：${res}</div>`;
    chat.scrollTop = chat.scrollHeight;
  }
}

async function fetchAi(apikey,model,task,prompt){
  let url="";
  let reqBody={model:"",messages:[]};
  if(model==="deepseek"){
    url="https://api.deepseek.com/v1/chat/completions";
    reqBody.model="deepseek-chat";
  }else if(model==="doubao"){
    url="https://ark.cn-beijing.volces.com/api/v3/chat/completions";
    reqBody.model="doubao-pro";
  }else{
    url="https://api.openai.com/v1/chat/completions";
    reqBody.model="gpt-3.5-turbo";
  }
  reqBody.messages=[
    {role:"system",content:prompt},
    {role:"user",content:task}
  ];
  try{
    let res = await fetch(url,{
      method:"POST",
      headers:{
        "Authorization":`Bearer ${apikey}`,
        "Content-Type":"application/json"
      },
      body:JSON.stringify(reqBody)
    })
    let data = await res.json();
    return data.choices?.[0]?.message?.content || "请求异常";
  }catch(e){
    return "接口请求失败，请检查密钥与网络";
  }
}
